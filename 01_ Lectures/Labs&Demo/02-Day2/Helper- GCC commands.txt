gcc -E program.c -o program.i  <-- -E: Stop after the preprocessing stage
				<-- -o program.i: Save the preprocessed output to a file


gcc -S program.i -o program.s   <-- -S: Stop after compilation; do not assemble.

gcc -c program.s -o program.o    <-- -c: Compile and assemble, but do not link.
